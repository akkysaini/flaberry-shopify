/* ---------- Availability & routing ---------- */
function eligible(pk,pin){
  const p=P[pk],PIN=PINS[pin]; if(!PIN) return [];
  const need=new Set([p.group,...p.cats.map(c=>CATS[c]?.group).filter(g=>g&&g!=="occasions")]);
  return PIN.v.filter(v=>{const V=VENDORS[v];return V&&[...need].every(g=>V.groups.includes(g))&&!p.cats.some(c=>V.ex.includes(c));});
}
function avail(pk,pin=S.pin){ if(!pin) return {known:false,ok:true,v:[]}; if(!PINS[pin]) return {known:true,ok:false,v:[],nopin:true}; const v=eligible(pk,pin); return {known:true,ok:v.length>0,v}; }
const visible=keys=>S.pin&&PINS[S.pin]?keys.filter(k=>avail(k).ok):keys;

/* ---------- Pricing ---------- */
const WEIGHTS=[["1/2 kg",1],["1 kg",1.85],["1.5 kg",2.7],["2 kg",3.5]];
function basePrice(p,opt={}){let b=p.price;if(isCake(p)){const w=WEIGHTS.find(w=>w[0]===opt.weight)||WEIGHTS[0];b=round9(b*w[1]);if(opt.egg==="Eggless")b+=50;}return b;}
function ruleFor(p,date){ if(!date) return null; return RULES.find(r=>r.active&&date>=r.from&&date<=r.to&&r.groups.some(g=>g===p.group||p.cats.some(c=>CATS[c]?.group===g))); }
function priceOn(p,opt,date){const b=basePrice(p,opt);const r=ruleFor(p,date);if(!r)return {price:b,base:b,rule:null};let v;if(r.useSaved&&p.vprice&&!isCake(p))v=Math.round(p.vprice*b/p.price);else v=round9(b*(1+r.uplift/100));return {price:Math.max(v,b),base:b,rule:r};}
